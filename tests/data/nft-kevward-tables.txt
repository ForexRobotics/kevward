table inet kevward {
	set allow4 {
		type ipv4_addr
		flags interval
		auto-merge
		elements = { 10.0.0.0/8, 192.168.0.0/16 }
	}

	set allow6 {
		type ipv6_addr
		flags interval
		auto-merge
	}

	set blocklist4 {
		type ipv4_addr
		flags timeout
		elements = { 1.2.3.4 timeout 14d expires 13d23h59m59s940ms, 5.6.7.8 timeout 14d expires 13d23h59m59s940ms }
	}

	set blocklist6 {
		type ipv6_addr
		flags timeout
	}

	set blockcidr4 {
		type ipv4_addr
		flags interval
		auto-merge
		elements = { 198.51.100.0/24 }
	}

	set blockcidr6 {
		type ipv6_addr
		flags interval
		auto-merge
	}

	chain input {
		type filter hook input priority mangle; policy accept;
		ip saddr @allow4 accept
		ip6 saddr @allow6 accept
		ip saddr @blocklist4 counter packets 0 bytes 0 drop
		ip6 saddr @blocklist6 counter packets 0 bytes 0 drop
		ip saddr @blockcidr4 counter packets 0 bytes 0 drop
		ip6 saddr @blockcidr6 counter packets 0 bytes 0 drop
	}
}
table inet kevward_harden {
	chain input {
		type filter hook input priority mangle - 10; policy accept;
		ip hdrlength != 5 counter packets 0 bytes 0 drop
	}
}
